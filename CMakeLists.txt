cmake_minimum_required(VERSION 3.20)
project(reqscope VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(REQSCOPE_WITH_CUPTI "Build the CUPTI collector when installed" ON)
option(REQSCOPE_BUILD_CUDA_FIXTURES "Build CUDA feasibility fixtures" OFF)
option(REQSCOPE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(REQSCOPE_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
find_package(Threads REQUIRED)
include(GNUInstallDirs)
add_library(reqscope_format STATIC format/json.cpp format/segment.cpp)
target_include_directories(reqscope_format PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(reqscope_analyzer STATIC analyzer/analyzer.cpp analyzer/records.cpp analyzer/clocks.cpp
  analyzer/linking.cpp analyzer/blame.cpp analyzer/report.cpp analyzer/request_report.cpp
  analyzer/explain.cpp analyzer/diff.cpp analyzer/perfetto.cpp)
target_link_libraries(reqscope_analyzer PUBLIC reqscope_format)
add_executable(reqscope cli/main.cpp cli/otel.cpp cli/client.cpp)
target_link_libraries(reqscope PRIVATE reqscope_analyzer)
file(RELATIVE_PATH REQSCOPE_INSTALLED_LIBRARY_RELATIVE_PATH
  "${CMAKE_INSTALL_FULL_BINDIR}" "${CMAKE_INSTALL_FULL_LIBDIR}/libreqscope.so")
target_compile_definitions(reqscope PRIVATE
  REQSCOPE_INSTALLED_LIBRARY_RELATIVE_PATH="${REQSCOPE_INSTALLED_LIBRARY_RELATIVE_PATH}")
add_library(reqscope_collector SHARED collector/collector.cpp collector/cupti_backend.cpp)
set_target_properties(reqscope_collector PROPERTIES OUTPUT_NAME reqscope)
target_link_libraries(reqscope_collector PRIVATE reqscope_format Threads::Threads ${CMAKE_DL_LIBS})
if(REQSCOPE_WITH_CUPTI)
  find_path(CUPTI_INCLUDE_DIR cupti.h HINTS /usr/local/cuda/include ENV CUPTI_INCLUDE_DIR)
  find_library(CUPTI_LIBRARY cupti HINTS /usr/local/cuda/lib64 ENV CUPTI_LIBRARY_DIR)
  if(CUPTI_INCLUDE_DIR AND CUPTI_LIBRARY)
    target_compile_definitions(reqscope_collector PRIVATE REQSCOPE_HAS_CUPTI=1)
    target_include_directories(reqscope_collector PRIVATE ${CUPTI_INCLUDE_DIR})
    target_link_libraries(reqscope_collector PRIVATE ${CUPTI_LIBRARY})
    get_filename_component(CUPTI_LIBRARY_DIRECTORY ${CUPTI_LIBRARY} DIRECTORY)
    set_target_properties(reqscope_collector PROPERTIES INSTALL_RPATH ${CUPTI_LIBRARY_DIRECTORY})
    message(STATUS "CUPTI backend: ${CUPTI_LIBRARY}")
  else()
    message(STATUS "CUPTI unavailable; manual collector and offline analysis remain available")
  endif()
endif()
if(REQSCOPE_BUILD_CUDA_FIXTURES)
  enable_language(CUDA)
  foreach(name streams copies graph_replay graph_mutation)
    add_executable(dag_${name} fixtures/dag/${name}.cu)
    set_target_properties(dag_${name} PROPERTIES CUDA_STANDARD 17)
    target_link_libraries(dag_${name} PRIVATE ${CMAKE_DL_LIBS})
  endforeach()
  target_compile_definitions(dag_graph_replay PRIVATE REQSCOPE_ENABLE_CONDITIONAL_GRAPHS=1)
endif()
install(TARGETS reqscope RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(TARGETS reqscope_collector LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES collector/reqscope.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/reqscope)
install(DIRECTORY docs/ DESTINATION ${CMAKE_INSTALL_DOCDIR}/docs)
install(FILES README.md DESTINATION ${CMAKE_INSTALL_DOCDIR})
install(DIRECTORY adapter/vllm/ DESTINATION ${CMAKE_INSTALL_DATADIR}/reqscope/adapter
  PATTERN "__pycache__" EXCLUDE PATTERN "*.egg-info" EXCLUDE
  PATTERN "build" EXCLUDE PATTERN "dist" EXCLUDE PATTERN "*.whl" EXCLUDE)
set(CPACK_GENERATOR TGZ)
set(CPACK_PACKAGE_NAME reqscope)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_SOURCE_GENERATOR TGZ)
set(CPACK_SOURCE_IGNORE_FILES
  "/[.]cache/;/[.]venv/;/[.]git/;/[.]aws/;/build[^/]*/;/install/;/packages/;/runs/;/bench-results/;/reqscope-(run|dump)-[^/]*/;/adapter/vllm/dist/;/__pycache__/;[.]egg-info/;[.]pyc$;[.]nsys-rep$;[.]sqlite$")
include(CPack)
enable_testing()
add_executable(reqscope_clock_probe_test tests/clock_probe_test.cpp)
target_include_directories(reqscope_clock_probe_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_test(NAME clock_probe_tests COMMAND reqscope_clock_probe_test)
add_executable(reqscope_record_text_test tests/record_text_test.cpp)
target_include_directories(reqscope_record_text_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(reqscope_record_text_test PRIVATE reqscope_format)
add_test(NAME record_text_tests COMMAND reqscope_record_text_test)
add_executable(reqscope_trace_sampler_test tests/trace_sampler_test.cpp)
target_include_directories(reqscope_trace_sampler_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_test(NAME trace_sampler_tests COMMAND reqscope_trace_sampler_test)
find_package(Python3 3.11 COMPONENTS Interpreter REQUIRED)
add_test(NAME independent_tests COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests -v)
set_tests_properties(independent_tests PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} ENVIRONMENT "REQSCOPE_BIN=$<TARGET_FILE:reqscope>;REQSCOPE_LIBRARY=$<TARGET_FILE:reqscope_collector>")
